\begin{frame}[t]{Antecedentes --- análisis de literatura}
    \footnotesize Revisión sistemática PRISMA 2020--2025 presentada como póster en \textbf{IEEE LARS/LARC 2025} (\cite{Trevino2025_LARSPoster}).
    \normalsize

    \vspace{0.15cm}
    Estrategias dominantes en motion control para escáneres de fotolitografía:
    \begin{itemize}
        \item \textbf{Control activo:} controladores no lineales tipo HIGS (\cite{Heertjes2025_NonlinearControl}).
        \item \textbf{Arquitecturas dinámicas:} triple etapas \textit{fine stages} (\cite{Al-Rawashdeh2022_Caracterization}).
        \item \textbf{\textit{Input shaping}:} trayectorias acotadas en \textit{jerk} (\cite{Al-Rawashdeh2023_Trajectories}).
    \end{itemize}
    \vspace{0.15cm}
    \begin{purpleblock}{Brecha}
        Los algoritmos optimizan métricas servo-mecánicas (MSE) \textbf{sin considerar el impacto morfológico} en el rendimiento (\textit{yield}) del IC. \emph{Pregunta de investigación:} ¿puede una función de costo informada por el patrón cerrar esta brecha sin penalizar productividad?
    \end{purpleblock}
\end{frame}

\begin{frame}{Hipótesis y objetivos}
    \begin{purpleblock}{Hipótesis}
        Una CNN entrenada sobre WM-811K, usada como función de costo, permite seleccionar parámetros cinemáticos de trayectorias de cuarto orden que \textbf{minimicen la probabilidad de defecto} sujeto a una \textbf{restricción de productividad}.
    \end{purpleblock}

    \vspace{0.1cm}
    \begin{greenblock}{Objetivo general}
        \textbf{Hallar perfiles de aceleración óptimos} para trayectorias \textit{step-and-scan} que mejoren el rendimiento estimado de la oblea sin penalizar el \textit{throughput}.
    \end{greenblock}

    \vspace{0.05cm}
    \textbf{Objetivos específicos:}
    \begin{enumerate}\setlength\itemsep{0pt}
        \item Generar trayectorias \textit{step-and-scan} acotadas en \textit{snap}.
        \item Entrenar una CNN sobre WM-811K para clasificación de defectos.
        \item Usar la CNN como función de costo para hallar los perfiles óptimos.
    \end{enumerate}
\end{frame}

\begin{frame}{Dinámica}
    \begin{itemize}
        \item Restringimos el movimiento a $f_{i_{j_x}}$, $f_{i_{j_y}}$ y la rotación planar $\theta_z$.
        \item Consideramos 16 cuerpos ($N_1=7$, $N_2=4$, $N_3=7$); 6 estados por cuerpo:
        \begin{equation}
            x =
            \begin{bmatrix}
                \underbrace{x_1, \ldots, x_6}_{\text{marco base (i=1)}} \;,\;
                \underbrace{x_7, \ldots, x_{6N_1}}_{\text{cadena 1}} \;,\;
                \underbrace{\ldots}_{\text{cadena 2}} \;,\;
                \underbrace{\ldots}_{\text{cadena 3}}
            \end{bmatrix}^T
        \end{equation}
        \item Para cuerpos internos:
        \begin{align}
            \ddot{\mathbf{x}}_{2p-1} &=
            \frac{1}{\bar{m}_{i_j}} \Big[ u^0_{i_j} - {}^0\bar{C}_{i_j}\dot{\mathbf{x}}_{2p-1} - {}^0\bar{K}_{i_j}\mathbf{x}_{2p-1} + {}^0\hat{C}_{(i+1)_j}\dot{\mathbf{x}}_{2p+1} + {}^0\hat{K}_{(i+1)_j}\mathbf{x}_{2p+1} \Big] \notag \\
            &\quad - \frac{1}{\bar{m}_{(i-1)_j}} \Big[ u^0_{(i-1)_j} - {}^0\bar{C}_{(i-1)_j}\dot{\mathbf{x}}_{2p-3} - {}^0\bar{K}_{(i-1)_j}\mathbf{x}_{2p-3} + {}^0\hat{C}_{i_j}\dot{\mathbf{x}}_{2p-1} + {}^0\hat{K}_{i_j}\mathbf{x}_{2p-1} \Big]
        \end{align}
    \end{itemize}
    \vspace{-0.2cm}
    {\scriptsize Casos frontera ($i{=}2$, $i{=}N_j$) detallados en el documento de tesis.}
\end{frame}
