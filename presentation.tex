\begin{frame}{Dinámica}
    \begin{center}\footnotesize
    $\underbrace{1_{\mathcal{F}_0} + \textstyle\sum_{j=1}^{3}(N_j{-}1)}_{=\,16\text{ cuerpos}}$ 
    $\ \big|\ $ 
    $N_1{=}7$ retícula, $N_2{=}4$ óptica, $N_3{=}7$ oblea 
    $\ \big|\ $ 
    6 estados/cuerpo $\Rightarrow x\in\mathbb{R}^{96}$
    \end{center}

    \begin{equation*}
        x =
        \big[\;
            \underbrace{x_1,\ldots,x_6}_{\text{base }\mathcal{F}_1} \;,\;
            \underbrace{x_7,\ldots,x_{42}}_{\text{retícula}} \;,\;
            \underbrace{x_{43},\ldots,x_{60}}_{\text{óptica}} \;,\;
            \underbrace{x_{61},\ldots,x_{96}}_{\text{oblea}}
        \;\big]^{\!T}
    \end{equation*}

    \begin{center}\footnotesize
    Por cuerpo global $p$: $(f_x, f_y, \theta_z, \dot f_x, \dot f_y, \dot\theta_z) \;\mapsto\; (x_{6p-5},\ldots,x_{6p})$
    \end{center}

    \vspace{0.1cm}
    {\footnotesize\textbf{Cuerpo intermedio} ($2{<}i{<}N_j$):}\vspace{-0.15cm}
    \begin{align*}
        \ddot{\mathbf{x}}_{2p-1} &=
        \tfrac{1}{\bar{m}_{i_j}} \Big[ u^0_{i_j} - {}^0\bar{C}_{i_j}\dot{\mathbf{x}}_{2p-1} - {}^0\bar{K}_{i_j}\mathbf{x}_{2p-1} + {}^0\hat{C}_{(i+1)_j}\dot{\mathbf{x}}_{2p+1} + {}^0\hat{K}_{(i+1)_j}\mathbf{x}_{2p+1} \Big] \\
        &\quad - \tfrac{1}{\bar{m}_{(i-1)_j}} \Big[ u^0_{(i-1)_j} - {}^0\bar{C}_{(i-1)_j}\dot{\mathbf{x}}_{2p-3} - {}^0\bar{K}_{(i-1)_j}\mathbf{x}_{2p-3} + {}^0\hat{C}_{i_j}\dot{\mathbf{x}}_{2p-1} + {}^0\hat{K}_{i_j}\mathbf{x}_{2p-1} \Big]
    \end{align*}

    \vspace{0.05cm}
    \begin{columns}[T,onlytextwidth]
    \begin{column}{0.50\textwidth}\scriptsize
        $i$: cuerpo en cadena \;$\cdot$\; $j$: cadena $\in\{1,2,3\}$\\[1pt]
        $p$: índice global \;$\cdot$\; ${}^{0}(\cdot)$: marco inercial $\mathcal{F}_0$\\[1pt]
        $u^0_{i_j}$: fuerza externa sobre $i_j$
    \end{column}
    \begin{column}{0.50\textwidth}\scriptsize
        $\bar m_{i_j}$: masa\\[1pt]
        $\bar C,\bar K$: enlace $(i{-}1){\to}i$\\[1pt]
        $\hat C,\hat K$: acoplamiento hacia abajo
    \end{column}
    \end{columns}
\end{frame}
